================================================================================
E-WASTE TFLITE VISION MODEL INTEGRATION GUIDE (FLUTTER)
================================================================================

1. OVERVIEW
--------------------------------------------------------------------------------
This guide details how to integrate the on-device MobileNetV3 TFLite model for 
offline scrap grading. The model takes a camera picture or gallery image, 
preprocesses it, and outputs classification probabilities across 6 scrap classes.


2. REQUIRED ASSET FILES
--------------------------------------------------------------------------------
Place both of these files in your Flutter assets directory:

  my_flutter_app/
  └── assets/
      └── models/
          ├── model.tflite
          └── labels.txt

Exact content of `labels.txt` (MUST match the index order 0 to 5):
--------------------------------------------------------------------------------
BATTERY_LITHIUM_PORTABLE
COPPER_HEAVY_INSULATED
LEAD_ACID
MOTHERBOARD_HIGH_GRADE
NON_RECYCLABLE
POWER_SUPPLY_LOW_GRADE
--------------------------------------------------------------------------------


3. PUBSPEC.YAML SETUP
--------------------------------------------------------------------------------
Add dependencies and register the assets:

dependencies:
  flutter:
    sdk: flutter
  tflite_flutter: ^0.10.4
  image: ^4.2.0

flutter:
  assets:
    - assets/models/model.tflite
    - assets/models/labels.txt


4. INPUT TENSOR & PREPROCESSING REQUIREMENTS
--------------------------------------------------------------------------------
* Input Tensor Shape : [1, 224, 224, 3]
* Data Type          : Float32
* Color Format       : RGB
* Normalization      : Standard PyTorch / ImageNet mean & std
    - Red Channel   : (pixel / 255.0 - 0.485) / 0.229
    - Green Channel : (pixel / 255.0 - 0.456) / 0.224
    - Blue Channel  : (pixel / 255.0 - 0.406) / 0.225


5. CONFIDENCE THRESHOLD & USER CONFIRMATION
--------------------------------------------------------------------------------
* Confidence Threshold: 0.70 (70%)

Rule:
- If max probability >= 0.70:
    Accept the classification automatically and calculate estimated payout.
- If max probability < 0.70:
    Display a confirmation dialog prompting the collector to manually verify 
    or override the category (especially between High-Grade and Low-Grade boards).


6. DART CLASSIFIER SERVICE CODE
--------------------------------------------------------------------------------
Save as `lib/services/ewaste_classifier.dart`:

```dart
import 'dart:io';
import 'dart:math' as math;
import 'package:flutter/services.dart';
import 'package:image/image.dart' as img;
import 'package:tflite_flutter/tflite_flutter.dart';

class PredictionResult {
  final String label;
  final double confidence;
  final bool requiresUserConfirmation;

  PredictionResult({
    required this.label,
    required this.confidence,
    required this.requiresUserConfirmation,
  });
}

class EWasteClassifier {
  Interpreter? _interpreter;
  List<String> _labels = [];
  bool _isReady = false;

  bool get isReady => _isReady;

  Future<void> initialize() async {
    try {
      // 1. Load model interpreter
      _interpreter = await Interpreter.fromAsset('assets/models/model.tflite');

      // 2. Load labels
      final rawLabels = await rootBundle.loadString('assets/models/labels.txt');
      _labels = rawLabels
          .split('\n')
          .map((line) => line.trim())
          .where((line) => line.isNotEmpty)
          .toList();

      _isReady = true;
      print("EWasteClassifier initialized with ${_labels.length} classes.");
    } catch (e) {
      print("Failed to initialize EWasteClassifier: $e");
    }
  }

  Future<PredictionResult?> classifyImage(File imageFile) async {
    if (!_isReady || _interpreter == null) {
      print("Classifier not initialized.");
      return null;
    }

    // 1. Read & decode image bytes
    final bytes = await imageFile.readAsBytes();
    final decoded = img.decodeImage(bytes);
    if (decoded == null) return null;

    // 2. Resize to 224x224
    final resized = img.copyResize(decoded, width: 224, height: 224);

    // 3. Preprocess to Float32 RGB normalized array [1, 224, 224, 3]
    var input = List.generate(
      1,
      (b) => List.generate(
        224,
        (y) => List.generate(224, (x) {
          final pixel = resized.getPixel(x, y);
          final r = (pixel.r / 255.0 - 0.485) / 0.229;
          final g = (pixel.g / 255.0 - 0.456) / 0.224;
          final bVal = (pixel.b / 255.0 - 0.406) / 0.225;
          return [r, g, bVal];
        }),
      ),
    );

    // 4. Output buffer [1, num_classes]
    var output = List.filled(1 * _labels.length, 0.0).reshape([1, _labels.length]);

    // 5. Run inference
    _interpreter!.run(input, output);

    List<double> rawScores = List<double>.from(output[0]);

    // 6. Apply Softmax to obtain true probabilities
    final probabilities = _softmax(rawScores);

    // 7. Find top class
    int topIndex = 0;
    double maxConfidence = -1.0;
    for (int i = 0; i < probabilities.length; i++) {
      if (probabilities[i] > maxConfidence) {
        maxConfidence = probabilities[i];
        topIndex = i;
      }
    }

    return PredictionResult(
      label: _labels[topIndex],
      confidence: maxConfidence,
      requiresUserConfirmation: maxConfidence < 0.70,
    );
  }

  List<double> _softmax(List<double> logits) {
    final maxLogit = logits.reduce(math.max);
    final expScores = logits.map((val) => math.exp(val - maxLogit)).toList();
    final sumExp = expScores.reduce((a, b) => a + b);
    return expScores.map((score) => score / sumExp).toList();
  }

  void dispose() {
    _interpreter?.close();
  }
}
